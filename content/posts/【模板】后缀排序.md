	rk[i]=str[i];

	for(int i=1;i<=len;++i){
		int p=sa[i].se;
		sa[i]={{rk[p],rk[p+w]},p};
	}
	sort(sa+1,sa+len+1);
	memcpy(oldrk,rk,sizeof(rk));
	for(int rks=0,i=1;i<=len;++i){
		if(sa[i].fi==sa[i-1].fi)

	}
}
